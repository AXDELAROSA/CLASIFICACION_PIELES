    Private Sub PM_Cargar()
        Try
            ListBox1.Items.Clear()
            PM_FO_TB_INIT(0)

            Dim VP_FOLIO As String = Lst_Folios.SelectedItem
            Dim VP_tabla As DataTable = Codigo_PEARL.FG_ACTION_EXECUTE_DATATABLE(VG_BD_DATA_02, "PG_SK_RP_SC_LOTE_X_TAG_NO", VP_FOLIO, True)

            ListBox4.Items.Clear()
            Dim row As DataRow
            For Each row In VP_tabla.Rows
                ListBox4.Items.Add(row.Item(0))
            Next

            PM_BT_CARGAR_LISTADO(Me, DGFolio, "PG_SK_RP_SC_X_TAG_NO")

            txt_pieles.Text = DGFolio.Rows.Count()
            txt_sqf.Text = Codigo_PEARL.FG_DG_COLUMN_SUM(VG_BD_DATA_02, DGFolio, "SQF")

            If DGFolio.Rows.Count() > 0 Then
                Dim VP_MOVEMENT As String = DGFolio.Rows(0).Cells("MOVEMENT").Value
                txt_Loc.Text = VP_MOVEMENT

                If VP_MOVEMENT = "" Then
                    txt_Loc.Text = "/"
                    Btn_Dvn.Enabled = False
                End If
            End If

            If Microsoft.VisualBasic.Left(txt_Loc.Text, 1) = "T" Then
                GroupBox3.Enabled = False
            Else
                GroupBox3.Enabled = True
                TB_FOL_ORDEN_DESTINO.Visible = True
                TB_FOL_ORDEN_DESTINO.Text = ""
                LBL_ORDEN_D.Visible = True
            End If

            VP_tabla.Clear()
            VP_tabla = Codigo_PEARL.FG_ACTION_EXECUTE_DATATABLE(VG_BD_DATA_02, "PG_SK_RP_FOLIO_X_TAG", VP_FOLIO, True)

            Dim VP_N_REGISTROS As Integer = VP_tabla.Rows.Count()

            If VP_N_REGISTROS > 0 Then
                Dim VP_ORDEN As String = (Codigo_PEARL.FG_DG_DATAROW_READ_STRING(VP_tabla.Rows(0), "JOBNO"))
                Dim VP_MACHINE As String = (Codigo_PEARL.FG_DG_DATAROW_READ_STRING(VP_tabla.Rows(0), "MACHINE"))
                Dim VP_STATUS As String = (Codigo_PEARL.FG_DG_DATAROW_READ_STRING(VP_tabla.Rows(0), "STATUS"))

                txt_orden.Text = VP_ORDEN
                If (txt_orden.Text = "      ") Then txt_orden.Text = "-"

                txt_Mesa.Text = VP_MACHINE
                If (txt_Mesa.Text = "          ") Then txt_Mesa.Text = "-"

                txt_tipo.Text = VP_STATUS
                If VP_STATUS = "B" Then Btn_Dvn.Enabled = False
            End If
        Catch ex As Exception
            Codigo_PEARL.PG_MESSAGE_ERROR_VS(0, "PM_Cargar")
        End Try
    End Sub

    Private Sub Button1_Click(sender As System.Object, e As System.EventArgs) Handles Btn_TransFolio.Click

        If DGFolio.SelectedRows.Count = 0 Then  'Compruebo si no hay ninguna piel seleccionada
            Codigo_PEARL.PG_MESSAGE_NOTICE(0, "Favor de seleccionar la Piel a transferir...")
            Exit Sub
        End If

        If txt_folio.Text = "" And Rb_exist.Checked = True And from_btn <> "Devolucion" Then ' Compruebo que tenga folio destino la transferencia
            Codigo_PEARL.PG_MESSAGE_NOTICE(0, "Favor de capturar el Folio destino...")
            txt_folio.Text = ""
            txt_folio.Focus()
            Exit Sub
        End If

        Try
            PM_LLENAR_DG2_CON_DGFolio_Seleccionado()

            If Rb_nuevo.Checked = True Then  'Transferir a un folio nuevo!
                If Trim(txt_tipo.Text) = "B" Then
                    If Not FM_Comprobar_CompEnFolio("FOLIO_NUEVO") Then  'Si el folio esta en MHI comprobar compatilidad, debido a que se encuentran todos los lotes mezclados
                        from_form = "Foliosvb"
                        Act.Show()
                        'Me.Close()
                        Exit Sub
                    End If
                End If
                '************************Transferir_nuevo**********************************
                Call PM_Transferir_nuevo()
                '**********************************************************
            Else
                Dim VP_TB_LOCACION As String = Trim(txt_Loc.Text)
                If VP_TB_LOCACION = "MQU" Then
                    Dim VP_MENSAJE As String = Codigo_PEARL.FG_ACTION_GetDato_OF_StoreProcedure(VG_BD_DATA_02, "PG_RN_VALIDA_USUARIO_MQU", cod_user, "MENSAJE")
                    If VP_MENSAJE <> "" Then
                        Codigo_PEARL.PG_MESSAGE_NOTICE(0, VP_MENSAJE)
                        Exit Sub
                    End If
                End If

                If DGFolio.SelectedRows.Count = 1 And txt_TransSQF.Text = "" Then
                    txt_TransSQF.Text = DGFolio.SelectedRows(0).Cells("SQF").Value
                End If

                If from_btn = "Devolucion" Then  'Si estamos realizando una devolucion de pieles a MHI (folio base)

                    Call PM_transferir_piel("DEVOLUCION")
                Else  'Si se trata de una transferencia a otro folio existente
                    Dim VP_FOLIO_DESTINO As String = Trim(txt_folio.Text)
                    Dim VP_FOLIO_ORIGEN As String = FM_SQL_PARAMETER_FOR_SEARCH()

                    If VP_FOLIO_DESTINO = VP_FOLIO_ORIGEN Then
                        Codigo_PEARL.PG_MESSAGE_NOTICE(0, "No es posible transferir dentro del mismo folio...")
                        txt_folio.SelectAll()
                        txt_folio.Focus()
                        Exit Sub
                    End If

                    Dim VP_tabla As DataTable = Codigo_PEARL.FG_ACTION_EXECUTE_DATATABLE(VG_BD_DATA_02, "PG_SK_RP_FOLIO_X_TAG", VP_FOLIO_DESTINO, True)

                    Dim VP_N_REGISTROS As Integer = VP_tabla.Rows.Count()
                    If VP_N_REGISTROS > 0 Then
                        Dim VP_STATUS As String = (Codigo_PEARL.FG_DG_DATAROW_READ_STRING(VP_tabla.Rows(0), "STATUS"))
                        'comentado para pruebas feg
                        If Trim(VP_STATUS) <> "B" Then
                            If Not FM_Comprobar_CompEnFolio("FOLIO_EXISTENTE") Then  'Si el folio esta o va para  MHI No comprobar compatilidad, debido a que se encuentran todos los lotes mezclados
                                from_form = "Foliosvb"
                                Exit Sub
                            End If
                        End If

                        from_btn = "Button1"
                        Call PM_transferir_piel("FOLIO_EXISTENTE")
                    Else
                        Codigo_PEARL.PG_MESSAGE_NOTICE(0, "Folio: <" + VP_FOLIO_DESTINO + " destino no encontrado...")
                        Exit Sub
                    End If
                End If
            End If

            from_btn = "Button1"
            from_form = "Foliosvb"
            Act.Show()

        Catch ex As Exception
            from_btn = "Button1"
            Codigo_PEARL.PG_MESSAGE_ERROR_VS(0, "Btn_TransFolio.Click")
        End Try
    End Sub

    Private Sub PM_LLENAR_DG2_CON_DGFolio_Seleccionado()
        If DataGridView2.Rows.Count > 0 Then
            DataGridView2.Rows.Clear()
        End If

        Try
            For Each VP_ROW As DataGridViewRow In DGFolio.SelectedRows
                Dim VP_INDEX As Integer = VP_ROW.Index
                Dim VP_LOTE_DGFOLIO As String = Trim(Codigo_PEARL.FG_DG_CELLS_READ(DGFolio, VP_INDEX, "Lote"))
                Dim VP_SQF_DGFOLIO As String = Trim(Codigo_PEARL.FG_DG_CELLS_READ(DGFolio, VP_INDEX, "SQF"))
                Dim VP_COLOR_DGFOLIO As String = Trim(Codigo_PEARL.FG_DG_CELLS_READ(DGFolio, VP_INDEX, "Color"))

                VP_LOTE_DGFOLIO = Trim(VP_LOTE_DGFOLIO).PadLeft(6, "0")

                For y As Integer = 0 To DataGridView2.Rows.Count - 1
                    Dim VP_LOTE_DG2 As String = Trim(Codigo_PEARL.FG_DG_CELLS_READ(DataGridView2, y, "LOT"))
                    Dim VP_SQF_DG2 As String = Trim(Codigo_PEARL.FG_DG_CELLS_READ(DataGridView2, y, "SQF"))

                    If VP_LOTE_DGFOLIO = VP_LOTE_DG2 Then
                        DataGridView2.Rows(y).Cells("SQF").Value = CDec(VP_SQF_DG2) + CDec(VP_SQF_DGFOLIO)
                        GoTo sig
                    End If
                Next

                DataGridView2.Rows.Add(VP_COLOR_DGFOLIO, "         " & VP_LOTE_DGFOLIO, Format(CDec(VP_SQF_DGFOLIO), "0.00"))
sig:
            Next
        Catch ex As Exception
            Codigo_PEARL.PG_MESSAGE_ERROR_VS(0, "PM_LLENAR_DG2_CON_DGFolio_Seleccionado")
        End Try
    End Sub

    Private Sub PM_transferir_piel(ByVal PP_TIPO_TRANSFERENCIA As String)
        Dim VP_FOLIO_DESTINO As Integer = 0
        Dim VP_FOLIO_ORIGEN As String = FM_SQL_PARAMETER_FOR_SEARCH()
        Dim VP_CB_COLOR As String = Trim(Codigo_PEARL.FG_COMBOBOX_ITEM_TEXT(CB_COLOR))
        Dim VP_TB_LOCACION As String = Trim(txt_Loc.Text)
        Dim VP_PRT_FORM_DEV_EXT As Integer = 0
        Dim VP_LOTE_PIEL_EXISTE As Integer = 0
        Dim VP_LOTE_PIEL_EXISTE_AGREGAR As Integer = 0
        Dim VP_LOCACION_DESTINO As String = ""

        Dim VP_TBL_PIEL_A_TRANSFERIR As New DataTable
        VP_TBL_PIEL_A_TRANSFERIR.Columns.Add("FOLIO")
        VP_TBL_PIEL_A_TRANSFERIR.Columns.Add("COLOR")

    Private Sub PM_Transferir_nuevo()

        Dim VP_TB_LOCACION As String = Trim(txt_Loc.Text)
        If VP_TB_LOCACION = "" Or VP_TB_LOCACION = "/" Then
            Codigo_PEARL.PG_MESSAGE_NOTICE(0, "Indique la Locacion origen de la transferencia....")
            txt_Loc.Enabled = True
            txt_Loc.Focus()
            Exit Sub
        End If

        If Not (Mid(VP_TB_LOCACION, 1, 1) = "T" Or VP_TB_LOCACION = "MHI" Or VP_TB_LOCACION = "MIT" Or VP_TB_LOCACION = "MQU" Or VP_TB_LOCACION = "MRD" Or VP_TB_LOCACION = "MCT" Or VP_TB_LOCACION = "MFP") Then   'Comprobamos que el origen de la transferencia sea valido
            Codigo_PEARL.PG_MESSAGE_NOTICE(0, "Proporcione una Locacion valida de la transferencia...")
            txt_Loc.Enabled = True
            txt_Loc.Focus()
            Exit Sub
        End If

        pos = 1
        If txt_orden.Text = "-" Then
            typef = "A"
        ElseIf txt_orden.Text <> "-" Then
            myValue = Trim(txt_orden.Text)
            typef = "C1"
        End If

        If Len(myValue) > 5 Then
            myValue = ""
        End If

        '--------------------------------------

        Dim VP_SELECT_TODOS As Integer = 0
        If DGFolio.SelectedRows.Count = DGFolio.Rows.Count Then
            VP_SELECT_TODOS = 1
        End If

        '--------OBTENGO LAS PIELES SELECCIONADAS---------
        Dim VP_LOTE_ARRAY As String = ""
        Dim VP_PIEL_ARRAY As String = ""
        Dim VP_SQF_ARRAY As String = ""

        For Each VP_ROW As DataGridViewRow In DGFolio.SelectedRows
            Dim VP_INDEX As Integer = VP_ROW.Index

            If VP_LOTE_ARRAY = "" Then
                VP_LOTE_ARRAY = Trim(Codigo_PEARL.FG_DG_CELLS_READ(DGFolio, VP_INDEX, "Lote"))
            Else
                VP_LOTE_ARRAY = VP_LOTE_ARRAY + "/" + Trim(Codigo_PEARL.FG_DG_CELLS_READ(DGFolio, VP_INDEX, "Lote"))
            End If

            If VP_PIEL_ARRAY = "" Then
                VP_PIEL_ARRAY = Trim(Codigo_PEARL.FG_DG_CELLS_READ(DGFolio, VP_INDEX, "Piel"))
            Else
                VP_PIEL_ARRAY = VP_PIEL_ARRAY + "/" + Trim(Codigo_PEARL.FG_DG_CELLS_READ(DGFolio, VP_INDEX, "Piel"))
            End If

            If VP_SQF_ARRAY = "" Then
                VP_SQF_ARRAY = Trim(Codigo_PEARL.FG_DG_CELLS_READ(DGFolio, VP_INDEX, "SQF"))
            Else
                VP_SQF_ARRAY = VP_SQF_ARRAY + "/" + Trim(Codigo_PEARL.FG_DG_CELLS_READ(DGFolio, VP_INDEX, "SQF"))
            End If

        Next
        '------------------------------------

        Dim VP_CB_COLOR As String = Trim(Codigo_PEARL.FG_COMBOBOX_ITEM_TEXT(CB_COLOR))
        Dim VP_FOLIO_LST As String = FM_SQL_PARAMETER_FOR_SEARCH()
        Dim VP_TB_ORDEN As String = Trim(txt_orden.Text)
        Dim VP_MAQUINA As String = Trim(txt_Mesa.Text)


        PM_LIMPIAR_VARIABLES_SQL_EJECUTAR()
        Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, VP_CB_COLOR)
        Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, VP_FOLIO_LST)
        Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, VP_TB_LOCACION, True)
        Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, VP_TB_ORDEN, True)
        Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, VP_MAQUINA, True)
        Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, VP_SELECT_TODOS)
        '------------------------------------
        Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, VP_LOTE_ARRAY, True)
        Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, VP_PIEL_ARRAY, True)
        Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, VP_SQF_ARRAY, True)

        Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, user, True)

        VM_EXITO = Codigo_PEARL.FG_ACTION_EXECUTE_NO_DATATABLE(VG_BD_DATA_02, Me, "PG_PM_TRANSFER_PIEL_A_FOLIO_NUEVO", VM_PARAMETROS, True, VM_CLAVE, VM_MENSAJE, False)

    Public Sub TextBox7_KeyPress(sender As Object, e As System.Windows.Forms.KeyPressEventArgs) Handles txt_search.KeyPress
        If Char.IsDigit(e.KeyChar) Then
            e.Handled = False
        ElseIf Char.IsControl(e.KeyChar) Then

        End If

                Dim VP_FOLIO As String = Trim(txt_search.Text)

                        Next

                        Lst_Folios.SelectedIndex = Lst_Folios.FindString(VP_FOLIO)
                        txt_search.Text = ""
                        txt_search.Focus()

                        If (Mid(txt_Loc.Text, 1, 1) = "T" Or Mid(txt_Mesa.Text, 1, 1) = "T") Then

    Private Sub PM_Comprobar_comp()
        ''*************************Comprobar compatibilidad de lotes al trasferir a una orden....*******************
        If RB_ORDEN.Checked = True And TB_LOC_ORDEN_DESTINO.Text <> "" Then
            Dim VP_COLOR_ORDEN As String = ""
            Dim VP_LOCACION_DESTINO As String = ""
            Dim VP_ORDEN As String = Trim(TB_LOC_ORDEN_DESTINO.Text)
            Dim VP_PARAMETROS As String = ""

            Codigo_PEARL.PG_CONTROL_PARAMETRO(VP_PARAMETROS, VP_ORDEN, True)

            Dim VP_tabla As DataTable = Codigo_PEARL.FG_ACTION_EXECUTE_DATATABLE(VG_BD_DATA_02, "PG_SK_CCJOBHDR_SQL_X_ORDEN", VP_PARAMETROS, True)

            If VP_tabla.Rows.Count > 0 Then
                VP_COLOR_ORDEN = Trim(Codigo_PEARL.FG_DG_DATAROW_READ_STRING(VP_tabla.Rows(0), "colour"))
                VP_COLOR_ORDEN = "F" + VP_COLOR_ORDEN

                VP_LOCACION_DESTINO = Trim(Codigo_PEARL.FG_DG_DATAROW_READ_STRING(VP_tabla.Rows(0), "Loc"))
            Else
                Codigo_PEARL.PG_MESSAGE_NOTICE(0, "La orden no Existe: " + VP_ORDEN + "...Transferencia Cancelada")
                Btn_TransLoc.Enabled = False
                Exit Sub
            End If

            Dim VP_COLOR_FOLIO As String = ""
            VP_COLOR_FOLIO = Trim(Codigo_PEARL.FG_DG_CELLS_READ(DGFolio, 0, "Color"))

            If VP_COLOR_FOLIO <> VP_COLOR_ORDEN Then
                Codigo_PEARL.PG_MESSAGE_NOTICE(0, "El color del Folio: " + VP_COLOR_FOLIO + " es diferende al de la Orden: " + VP_COLOR_ORDEN + " ..Transferencia Cancelada")
                Btn_TransLoc.Enabled = False
                Exit Sub
            End If

            ListBox1.Items.Clear()
            VP_tabla.Clear()
            PM_LIMPIAR_VARIABLES_SQL_EJECUTAR()

            Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, VP_COLOR_FOLIO, True)
            Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, VP_LOCACION_DESTINO, True)
            Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, VP_ORDEN, True)

            VP_tabla = Codigo_PEARL.FG_ACTION_EXECUTE_DATATABLE(VG_BD_DATA_02, "PG_LI_IMLSTRX_SQL_Y_IMINVTRX_SQL_X_COLOR_LOCACION_Y_ORDEN", VM_PARAMETROS, True)

            If VP_tabla.Rows.Count > 0 Then
                For row_tbl As Integer = 0 To VP_tabla.Rows.Count - 1
                    Dim VP_SER_LOT_NO As String = (Codigo_PEARL.FG_DG_DATAROW_READ_STRING(VP_tabla.Rows(row_tbl), "ser_lot_no"))
                    thislot = VP_SER_LOT_NO

                    Dim VP_TRX_QTY As String = (Codigo_PEARL.FG_DG_DATAROW_READ_STRING(VP_tabla.Rows(row_tbl), "trx_qty"))
                    Dim VP_LEV_NO As Integer = (Codigo_PEARL.FG_DG_DATAROW_READ_INT(VP_tabla.Rows(row_tbl), "lev_no"))
                    Dim VP_DOC_TYPE As String = (Codigo_PEARL.FG_DG_DATAROW_READ_STRING(VP_tabla.Rows(row_tbl), "doc_type"))

                    Select Case VP_DOC_TYPE
                        Case "I"
                            qtyallocated = qtyallocated - CDbl(VP_TRX_QTY)
                        Case "R"
                            qtyallocated = qtyallocated + CDbl(VP_TRX_QTY)
                        Case "T"
                            If VP_LEV_NO = 1 Then
                                qtyallocated = qtyallocated + CDbl(VP_TRX_QTY)
                            Else
                                qtyallocated = qtyallocated - CDbl(VP_TRX_QTY)
                            End If
                        Case Else
                            'other transaction types ignored
                    End Select

                    'build allocated qty
                    'insert into listbox
                    If qtyallocated > 0 Then

                        Dim VP_SER_LOT_NO_SIG As String = ""
                        If VP_tabla.Rows.Count = (row_tbl + 1) Then
                            VP_SER_LOT_NO_SIG = 0
                        Else
                            VP_SER_LOT_NO_SIG = (Codigo_PEARL.FG_DG_DATAROW_READ_STRING(VP_tabla.Rows(row_tbl + 1), "ser_lot_no"))
                        End If

                        If thislot <> VP_SER_LOT_NO_SIG Then
                            ListBox1.Items.Add(thislot.PadLeft(6, "0"))
                            qtyallocated = 0
                        End If
                    End If
                Next
            End If
            '''''este bloque comprueba los compatibles.
            Dim VP_TBL_LOTE_NO_COMPATIBLE As New DataTable
            VP_TBL_LOTE_NO_COMPATIBLE.Columns.Add("LOTE_A")
            VP_TBL_LOTE_NO_COMPATIBLE.Columns.Add("LOTE_B")
            VP_TBL_LOTE_NO_COMPATIBLE.Columns.Add("MENSAJE")

            For lst As Integer = 0 To ListBox4.Items.Count - 1
                For y As Integer = 0 To ListBox1.Items.Count - 1
                    If Trim(ListBox4.Items(lst)) = Trim(ListBox1.Items(y)) Then
                        GoTo sig
                    End If

                    Dim VP_SER_LOT_NO_1 As String = Trim(ListBox1.Items(y))
                    Dim VP_SER_LOT_NO_2 As String = Trim(ListBox4.Items(lst))

                    PM_LIMPIAR_VARIABLES_SQL_EJECUTAR()
                    Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, VP_SER_LOT_NO_1, True)
                    Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, VP_SER_LOT_NO_2, True)

                    VP_tabla.Clear()
                    VP_tabla = Codigo_PEARL.FG_ACTION_EXECUTE_DATATABLE(VG_BD_DATA_02, "PG_SK_SER_LOT_COMP_SQL_X_LOTE", VM_PARAMETROS, True)

                    If VP_tabla.Rows.Count > 0 Then
                        Dim VP_COMP As String = (Codigo_PEARL.FG_DG_DATAROW_READ_STRING(VP_tabla.Rows(0), "comp"))

                        If VP_COMP <> "Y" Then
                            VP_TBL_LOTE_NO_COMPATIBLE.Rows.Add(VP_SER_LOT_NO_1, VP_SER_LOT_NO_2, "No se a comprobado la compatibilidad")
                        Else
                            GoTo sig
                        End If
                    Else
                        Codigo_PEARL.PG_MESSAGE_NOTICE(0, "Aun no se a comprobado la compatibilidad con el Lote: " + VP_SER_LOT_NO_1 + "...Transferencia Cancelada")
                        Btn_TransLoc.Enabled = False
                        'CB_LOC.SelectedIndex = CB_LOC.FindString(Trim("MHI"))
                        Exit Sub
                    End If
sig:
                Next
            Next
            ListBox1.Items.Clear()

            If VP_TBL_LOTE_NO_COMPATIBLE.Rows.Count > 0 Then

    Function FM_Comprobar_CompEnFolio(ByVal PP_TIPO_COMPROBACION As String) As Boolean
        ''********************Comprobar compatibilidad de lotes agregados a un mismo folio....*******************
        PM_LLENAR_DG2_CON_DGFolio_Seleccionado()

        Dim VP_TBL_LOTE_EXISTENTE As New DataTable
        Dim VP_COLOR As String = Trim(Codigo_PEARL.FG_DG_CELLS_READ(DGFolio, 0, "Color"))

        If PP_TIPO_COMPROBACION = "FOLIO_EXISTENTE" Then

            Dim VP_LOCACION As String = Codigo_PEARL.FG_ACTION_GetDato_OF_StoreProcedure(VG_BD_DATA_02, "PG_GET_VALOR_LOCACION", Trim(txt_folio.Text), "LOCACION")

            PM_LIMPIAR_VARIABLES_SQL_EJECUTAR()
            Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, VP_COLOR, True)
            Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, Trim(VP_LOCACION), True)
            Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, Trim(TB_FOL_ORDEN_DESTINO.Text), True)

            VP_TBL_LOTE_EXISTENTE = Codigo_PEARL.FG_ACTION_EXECUTE_DATATABLE(VG_BD_DATA_02, "PG_GET_IMINVTRX_SQL_LOTE_X_COLOR_LOCACION_ORDEN", VM_PARAMETROS, True)
        Else
            VP_TBL_LOTE_EXISTENTE.Columns.Add("LOTE")
        End If

        If DataGridView2.Rows.Count < 2 And VP_TBL_LOTE_EXISTENTE.Rows.Count < 1 Then
            Return True
        Else
            For y As Integer = 0 To DataGridView2.Rows.Count - 1
                Dim VP_LOTE As String = Trim(DataGridView2.Rows(y).Cells("LOT").Value)

                Dim VP_N_ROW() As DataRow
                VP_N_ROW = VP_TBL_LOTE_EXISTENTE.Select("LOTE='" + VP_LOTE + "'")

                If VP_N_ROW.Length < 1 Then
                    VP_TBL_LOTE_EXISTENTE.Rows.Add(VP_LOTE)
                End If
            Next
        End If

        Dim VP_TBL_LOTE_NO_COMPATIBLE As New DataTable
        VP_TBL_LOTE_NO_COMPATIBLE.Columns.Add("LOTE_A")
        VP_TBL_LOTE_NO_COMPATIBLE.Columns.Add("LOTE_B")
        VP_TBL_LOTE_NO_COMPATIBLE.Columns.Add("MENSAJE")

        For row_x As Integer = 0 To VP_TBL_LOTE_EXISTENTE.Rows.Count - 1
            For row_y As Integer = 0 To VP_TBL_LOTE_EXISTENTE.Rows.Count - 1

                Dim VP_LOT_X As String = Trim(Codigo_PEARL.FG_DG_DATAROW_READ_STRING(VP_TBL_LOTE_EXISTENTE.Rows(row_x), "LOTE"))
                Dim VP_LOT_Y As String = Trim(Codigo_PEARL.FG_DG_DATAROW_READ_STRING(VP_TBL_LOTE_EXISTENTE.Rows(row_y), "LOTE"))

                If VP_LOT_X = VP_LOT_Y Then
                    GoTo sig
                End If

                PM_LIMPIAR_VARIABLES_SQL_EJECUTAR()
                Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, VP_LOT_Y, True)
                Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, VP_LOT_X, True)

                Dim VP_tabla_Com As DataTable = Codigo_PEARL.FG_ACTION_EXECUTE_DATATABLE(VG_BD_DATA_02, "PG_SK_SER_LOT_COMP_SQL_X_LOTE", VM_PARAMETROS, True)

                If VP_tabla_Com.Rows.Count > 0 Then
                    For row_tbl As Integer = 0 To VP_tabla_Com.Rows.Count - 1
                        Dim VP_COMP As String = (Codigo_PEARL.FG_DG_DATAROW_READ_STRING(VP_tabla_Com.Rows(row_tbl), "comp"))

                        If VP_COMP <> "Y" Then
                            VP_TBL_LOTE_NO_COMPATIBLE.Rows.Add(VP_LOT_X, VP_LOT_Y, "Aun no se a comprobado la compatibilidad")
                        Else
                            GoTo sig
                        End If
                    Next
                Else
                    Codigo_PEARL.PG_MESSAGE_NOTICE(0, "Aun no se a comprobado la compatibilidad del Lote: " + VP_LOT_X + " y el Lote: " + VP_LOT_Y + "")
                    Btn_TransLoc.Enabled = False
                    Return False
                End If
sig:
            Next
        Next

        If VP_TBL_LOTE_NO_COMPATIBLE.Rows.Count > 0 Then
            Dim VP_FORMA As New FO_AUTORIZAR_COMPATIBLE()
            VP_FORMA.PM_FO_SHOW(VP_TBL_LOTE_NO_COMPATIBLE, "Compatibles")
            VP_FORMA.ShowDialog()

            If VP_FORMA.Tag = "OK" Then
                Return True
            Else
                Return False
            End If
        Else
            Return True
        End If
    End Function

            If VP_TB_LOCACION <> "MQU" Then
                Codigo_PEARL.PG_MESSAGE_NOTICE(0, "Solo los Folios en MQU pueden ser borrados...")
                Exit Sub
            End If
            '-------------------------------------------------------------

            PM_LIMPIAR_VARIABLES_SQL_EJECUTAR()
            Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, VP_FOLIO_ORIGEN)
            Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, VP_TB_LOCACION, True)
            Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, txt_comentarios, True)
            Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, user, True)
            Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, My.Computer.Name, True)

            VM_EXITO = Codigo_PEARL.FG_ACTION_EXECUTE_NO_DATATABLE(VG_BD_DATA_02, Me, "PG_PM_TRANSFER_ISSUE", VM_PARAMETROS, True, VM_CLAVE, VM_MENSAJE, False)

            If VM_EXITO Then
                Codigo_PEARL.PG_MESSAGE_NOTICE(0, "Operacion exitosa...")
                PM_LIMPIAR_VARIABLES_SQL_EJECUTAR()
                Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, "B", True) ' ESTATUS
                Codigo_PEARL.PG_CONTROL_PARAMETRO(VM_PARAMETROS, Trim(CB_COLOR.Text), True)
                Dim VP_FOLIO_DESTINO As String = Codigo_PEARL.FG_ACTION_GetDato_OF_StoreProcedure(VG_BD_DATA_02, "PG_GET_FOLIO_DESTINO", VM_PARAMETROS, "FOLIO_DESTINO")

                txt_search.Text = VP_FOLIO_DESTINO
                txt_search.Focus()
                SendKeys.Send("{ENTER}")
            Else
                Codigo_PEARL.PG_MESSAGE_NOTICE(0, VM_MENSAJE)
            End If
        Catch ex As Exception
            Codigo_PEARL.PG_MESSAGE_ERROR_VS(0, "PM_BT_ISSUE")
        End Try
    End Sub

    Private Sub Button15_Click(sender As System.Object, e As System.EventArgs) Handles Btn_Sel.Click
        Dim sel As Boolean = False
        Dim total As Decimal = 0
        Dim cont As Integer = 0
        DGFolio.ClearSelection()

        Try
            ' Display message, title, and default value.
            Sel_folio.Tag = "cancel"
            Sel_folio.ShowDialog()

            If Sel_folio.Tag = "cancel" Then
                folio_sel = ""
            End If

            Try
                Sel_folio.Dispose()
            Catch ex As Exception
            End Try

            If folio_sel <> "" Then
                Dim VP_tabla As DataTable = Codigo_PEARL.FG_ACTION_EXECUTE_DATATABLE(VG_BD_DATA_02, "PG_SK_RP_SC_TEMP_X_TAG_NO", folio_sel, True)

                If VP_tabla.Rows.Count > 0 Then
                    For row_tbl As Integer = 0 To VP_tabla.Rows.Count - 1
                        Dim VP_TBL_HIDE As String = (Codigo_PEARL.FG_DG_DATAROW_READ_STRING(VP_tabla.Rows(row_tbl), "hide"))
                        Dim VP_TBL_SQF As String = (Codigo_PEARL.FG_DG_DATAROW_READ_STRING(VP_tabla.Rows(row_tbl), "sqf"))
                        Dim VP_TBL_LOT As String = (Codigo_PEARL.FG_DG_DATAROW_READ_STRING(VP_tabla.Rows(row_tbl), "lot"))

                        For row_dg As Integer = 0 To DGFolio.Rows.Count - 1
                            Dim VP_DGFOLIO_HIDE As String = Codigo_PEARL.FG_DG_CELLS_READ(DGFolio, row_dg, "PIEL")
                            Dim VP_DGFOLIO_SQF As String = Codigo_PEARL.FG_DG_CELLS_READ(DGFolio, row_dg, "SQF")
                            Dim VP_DGFOLIO_LOT As String = Codigo_PEARL.FG_DG_CELLS_READ(DGFolio, row_dg, "LOTE")

                            If Trim(VP_DGFOLIO_HIDE) = "0754" Then
                                sel = sel
                            End If

                            If (Trim(VP_TBL_HIDE) = Trim(VP_DGFOLIO_HIDE)) And (Trim(VP_TBL_SQF) = Trim(VP_DGFOLIO_SQF)) And (Trim(VP_TBL_LOT) = Trim(VP_DGFOLIO_LOT)) Then
                                cont = cont + 1
                                total = total + VP_TBL_SQF
                                DGFolio.Rows(row_dg).Selected = True
                                sel = True
                                Exit For
                            End If
                        Next

                        If sel = False Then
                            Codigo_PEARL.PG_MESSAGE_NOTICE(0, " Piel: <" + VP_TBL_HIDE + "> NO encontrada!")
                            Exit Sub
                        End If

                        sel = False
                    Next
                Else
                    Codigo_PEARL.PG_MESSAGE_NOTICE(0, " Folio: <" + folio_sel + "> NO encontrado...!")
                    folio_sel = ""
                    Exit Sub
                End If
                Codigo_PEARL.PG_MESSAGE_NOTICE(0, " Pieles Seleccionadas: <" + cont.ToString + "> Pies Cuadrados: <" + total.ToString + ">!")
            End If
        Catch ex As Exception
            Codigo_PEARL.PG_MESSAGE_ERROR_VS(0, "Btn_Sel.Click")
        End Try
    End Sub
